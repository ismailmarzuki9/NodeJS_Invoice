CREATE TABLE invoice (
    id_in VARCHAR PRIMARY KEY,
    no INTEGER,
    description TEXT,
    tic_number VARCHAR(50),
    nominal NUMERIC(15,2),
    total NUMERIC(15,2),
    date_issuance VARCHAR(50),
    Due_Date VARCHAR(50),
    nama_instansi VARCHAR(25),
    alamat VARCHAR(50),
    teruntuk VARCHAR(20),
    tggl_data_masuk TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    tggl_data_terupdate TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE TABLE users (
    id  SERIAL PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
